    function cleanWordLookup(text) {
      if (!text) return '';
      return text.replace(/[\u060C\u061B\u061F\u06D4.,:;()[\]{}«»"'—!؟?،ـ]/g, '').trim();
    }

    function selectWord(el, rawWord, trans) {
      document.querySelectorAll('.ar-word.user-selected').forEach(w => w.classList.remove('user-selected'));
      if (el) el.classList.add('user-selected');

      const clean = cleanWordLookup(rawWord);
      const fullVocalized = (el && el.getAttribute('data-tashkeel')) ? el.getAttribute('data-tashkeel') : rawWord;
      if (inspAr) inspAr.textContent = fullVocalized;
      if (inspRu) inspRu.textContent = trans || (el ? el.getAttribute('data-trans') : '') || '—';
      currentSelectedWord = clean || rawWord;

      if (inspBar) inspBar.style.transform = 'translateY(0)';

      playWordAudio(rawWord);
    }

    function playWordAudio(rawWord) {
      if (!rawWord) return;
      const word = rawWord.trim();
      const cleaned = cleanWordLookup(word);

      let file = null;

      // 1. Direct check for Salawat
      if (word === 'ﷺ' || cleaned === 'ﷺ' || word.includes('ﷺ') || word.includes('صَلَّى') || cleaned.includes('صلى')) {
        file = 'salawat_human.mp3';
      }
      // 2. Direct check for Radiyallahu anhu / anha
      else if (word.includes('عَنْهَا') && (word.includes('رَضِيَ') || word.includes('رضي'))) {
        file = 'radiyallahu_anha.mp3';
      } else if (word.includes('عَنْهُ') && (word.includes('رَضِيَ') || word.includes('رضي'))) {
        file = 'radiyallahu_anhu.mp3';
      }
      // 3. Direct check for Allah (recited by Sheikh Mahmoud Khalil Al-Husary)
      else if (cleaned.replace(/[\u064B-\u065F\u0670\u0671]/g, '').replace(/[ٱإأآ]/g, 'ا') === 'الله') {
        if (cleaned.includes('ِ') || word.includes('ِ') || word.includes('هِ')) {
          file = 'h_allah_kasra.mp3';
        } else if (cleaned.includes('ُ') || word.includes('ُ') || word.includes('هُ')) {
          file = 'h_allah_damma.mp3';
        } else if (cleaned.endsWith('َ') || word.includes('هَ')) {
          file = 'h_allah_fatha.mp3';
        } else {
          file = 'h_allah_damma.mp3';
        }
      }
      // 4. General dictionary lookup
      else {
        file = WORDS_AUDIO_MAP[word] || WORDS_AUDIO_MAP[cleaned];
      }

      if (file) {
        if (wordAudioPlayer) {
          wordAudioPlayer.pause();
        }
        wordAudioPlayer = new Audio('audio/words/' + file);
        wordAudioPlayer.play().catch(e => {
          wordAudioPlayer = new Audio('audio/' + file);
          wordAudioPlayer.play().catch(err => console.log('Audio error:', err));
        });
      } else {
        // Web Speech Fallback if available
        if ('speechSynthesis' in window) {
          window.speechSynthesis.cancel();
          const utterance = new SpeechSynthesisUtterance(cleaned || word);
          utterance.lang = 'ar-SA';
          utterance.rate = 0.85;
          window.speechSynthesis.speak(utterance);
        }
      }
    }

    function formatTime(seconds) {
      if (isNaN(seconds)) return '0:00';
      const m = Math.floor(seconds / 60);
      const s = Math.floor(seconds % 60);
      return `${m}:${s < 10 ? '0' : ''}${s}`;

            // Allow a graceful hold of up to 100ms or until the next item starts
            const maxHoldMs = nextTc ? Math.min(tc.endMs + 100, nextTc.startMs) : tc.endMs + 100;

            if (effectiveMs >= tc.startMs && effectiveMs <= maxHoldMs) {
              if (tc.isQuran) {
                matchedQuranIdx = tc.quranIndex !== undefined ? tc.quranIndex : 0;
              } else {
                matchedWordIdx = tc.wordIndex;
              }
              break;
            }
          }

          if (matchedQuranIdx >= 0) {
            // Quran ayah is reciting: illuminate Quran verse badge and clear regular word highlights
            words.forEach(w => w.classList.remove('audio-active'));
            let activeQuran = art.querySelector(`.quran-inline[data-quran-idx="${matchedQuranIdx}"]`);
            if (!activeQuran && matchedQuranIdx < quranEls.length) {
              activeQuran = quranEls[matchedQuranIdx];
            }
            quranEls.forEach(q => {
              if (q === activeQuran) {
                if (!q.classList.contains('audio-active')) {
                  q.classList.add('audio-active');
                  q.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'nearest' });
                }
              } else {
                q.classList.remove('audio-active');
              }
            });
          } else if (matchedWordIdx >= 0) {
            // Regular word is being spoken: highlight word and clear Quran verse highlight
            quranEls.forEach(q => q.classList.remove('audio-active'));
            let activeEl = art.querySelector(`.ar-word[data-idx="${matchedWordIdx}"]`);
            if (!activeEl && matchedWordIdx < words.length) {
              activeEl = words[matchedWordIdx];
            }
            words.forEach(w => {
              if (w === activeEl) {
                if (!w.classList.contains('audio-active')) {
                  w.classList.add('audio-active');
                  w.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'nearest' });
                }
              } else {
                w.classList.remove('audio-active');
              }
            });
          } else {
            // In pause or silence between sentences/quotes
            words.forEach(w => w.classList.remove('audio-active'));
            quranEls.forEach(q => q.classList.remove('audio-active'));
          }
        } else {
          // Linear fallback
          const pct = paraAudio.currentTime / (paraAudio.duration || 1);
          const currentIdx = Math.min(words.length - 1, Math.floor(pct * words.length));
          words.forEach((w, idx) => {
            if (idx === currentIdx) {
              w.classList.add('audio-active');
            } else {
              w.classList.remove('audio-active');
            }
          });
        }
      }

      paraAudio.ontimeupdate = syncKaraoke;

      paraAudio.play().then(() => {
        if (btn) {
          const icon = btn.querySelector('.play-icon');
          const txt = btn.querySelector('.play-text');
          if (icon) icon.textContent = '⏸';
          if (txt) txt.textContent = 'Пауза';
          btn.classList.add('playing');
        }
        if (card) card.classList.add('playing');
        activeParagraphAudio = paraAudio;
        activeParaNum = num;
        syncTimer = setInterval(syncKaraoke, 20);
      }).catch(e => console.error('Audio play error:', e));

      paraAudio.onended = () => {
        if (syncTimer) {
          clearInterval(syncTimer);
          syncTimer = null;
        }
        if (btn) {
          const icon = btn.querySelector('.play-icon');
          const txt = btn.querySelector('.play-text');
          if (icon) icon.textContent = '▶';
          if (txt) txt.textContent = 'Слушать абзац';
          btn.classList.remove('playing');
        }
        if (card) card.classList.remove('playing');
        clearActiveWords();
        activeParagraphAudio = null;
        activeParaNum = null;
      };
    }

          if (left + tooltip.offsetWidth > window.innerWidth - 10) {
            left = Math.max(10, window.innerWidth - tooltip.offsetWidth - 10);
          }
          tooltip.style.left = left + 'px';
          tooltip.style.top = Math.max(10, (rect.top + window.scrollY - tooltip.offsetHeight - 8)) + 'px';
        }

        const vocalized = word.getAttribute('data-tashkeel') || word.textContent.trim();
        inspAr.textContent = vocalized;
        inspRu.textContent = trans || '—';
        currentSelectedWord = vocalized;
      });

      word.addEventListener('mouseleave', () => {

    // Font family change (Amiri, Scheherazade New, Amiri Quran)
    function changeArabicFont(fontFamily) {
      if (!fontFamily) return;
      document.documentElement.style.setProperty('--arabic-font', fontFamily);
      document.querySelectorAll('.arabic-article, .ar-word, .arabic-title, .para-title, .verb-ar, .vocab-ar, .inspector-ar').forEach(el => {
        el.style.fontFamily = fontFamily;
      });
      localStorage.setItem('site_ar_font', fontFamily);
      const sel = document.getElementById('fontSelect');
      if (sel) sel.value = fontFamily;
    }

    // Font size control
    let currentFontSizeRem = parseFloat(localStorage.getItem('site_font_size')) || 1.65;

    function applyFontSize() {
      document.querySelectorAll('.arabic-article').forEach(el => {
        el.style.fontSize = currentFontSizeRem + 'rem';
        el.style.lineHeight = '2.85';
      });
      localStorage.setItem('site_font_size', currentFontSizeRem);
    }

    function adjustFontSize(step) {
      const delta = (step > 0 ? 0.15 : -0.15);
      currentFontSizeRem = Math.max(1.1, Math.min(2.8, Math.round((currentFontSizeRem + delta) * 100) / 100));
      applyFontSize();
    }

    function changeFontSize(delta) {
      currentFontSizeRem = Math.max(1.1, Math.min(2.8, currentFontSizeRem + delta));
      applyFontSize();
    }

      isQuranFont = !isQuranFont;

      changeArabicFont(fontName);

      isTashkeelOn = !isTashkeelOn;
      localStorage.setItem('site_tashkeel', isTashkeelOn ? 'true' : 'false');

    function applyTashkeel() {
      const btn = document.getElementById('tashkeelBtn');
      if (btn) {
        btn.textContent = isTashkeelOn ? '🔤 Ташкиль: Вкл' : '🔤 Ташкиль: Выкл';
        if (isTashkeelOn) {
          btn.classList.add('btn-active');
        } else {
          btn.classList.remove('btn-active');
        }
      }
      document.querySelectorAll('.ar-word, .para-title').forEach(el => {
        const plain = el.getAttribute('data-plain');
        const full = el.getAttribute('data-tashkeel');
        if (plain && full) {
          el.textContent = isTashkeelOn ? full : plain;
        }
      });
    }

    // Theme (Dark / Light)
    function toggleTheme() {
      const cur = document.documentElement.getAttribute('data-theme') || document.body.getAttribute('data-theme') || 'light';
      const target = cur === 'dark' ? 'light' : 'dark';
      document.documentElement.setAttribute('data-theme', target);
      document.body.setAttribute('data-theme', target);
      localStorage.setItem('site_theme', target);
      const btn = document.getElementById('themeBtn');
      if (btn) btn.textContent = target === 'dark' ? '☀️ Светлая' : '🌙 Тёмная';
    }

    (function initSavedSettings() {
